model Job {
  id                   String              @id @default(uuid())
  jobNumber            String              @unique
  customerId           String
  customer             Customer            @relation(fields: [customerId], references: [id])
  careOfPartyId        String?
  careOfParty          CareOfParty?        @relation(fields: [careOfPartyId], references: [id])
  manualJobNumber      String?
  jobType              String              // installation, maintenance, repair, inspection
  remarks              String?
  /// JSON array of completion proof photos (data URLs or https URLs) from field tech
  completionPhotos     String?
  status               String              @default("Created") // Created, Assigned, Accepted, InProgress, Paused, CompletedPendingVerification, Finalized, Verified
  assignedTechnicianId String?
  assignedTechnician   Employee?           @relation(fields: [assignedTechnicianId], references: [id])
  discountAmount       Float               @default(0)
  discountReason       String?
  qualityFlag          String?             // clean, disputed
  verifiedChecklist    String?             // JSON of checked items
  createdAt            DateTime            @default(now())
  finalizedAt          DateTime?
  verifiedAt           DateTime?
  items                JobItem[]
  statusHistory        JobStatusHistory[]
  inventoryRequests    InventoryRequest[]
  stockReturns         StockReturn[]
  expenseClaims        JobExpenseClaim[]
  hisaabSettlements    HisaabSettlement[]
  feedbackCalls        FeedbackCall[]
  locationPings        TechnicianLocationPing[]
}

model JobItem {
  id              String  @id @default(uuid())
  jobId           String
  job             Job     @relation(fields: [jobId], references: [id], onDelete: Cascade)
  description     String
  quantityPlanned Float
  quantityActual  Float?
  unitRate        Float
}

model JobStatusHistory {
  id        String   @id @default(uuid())
  jobId     String
  job       Job      @relation(fields: [jobId], references: [id], onDelete: Cascade)
  fromStatus String
  toStatus   String
  changedBy  String
  changedAt  DateTime @default(now())
  metaJson   String?  // JSON string
}

model InventoryRequest {
  id           String   @id @default(uuid())
  jobId        String
  job          Job      @relation(fields: [jobId], references: [id], onDelete: Cascade)
  technicianId String
  item         String
  qtyRequested Float
  status       String   @default("pending") // pending, issued, rejected
  createdAt    DateTime @default(now())
}

model StockReturn {
  id             String    @id @default(uuid())
  jobId          String
  job            Job       @relation(fields: [jobId], references: [id], onDelete: Cascade)
  technicianId   String
  item           String
  qtyReturned    Float
  acknowledgedBy String?
  acknowledgedAt DateTime?
  createdAt      DateTime  @default(now())
}

model JobExpenseClaim {
  id           String    @id @default(uuid())
  jobId        String
  job          Job       @relation(fields: [jobId], references: [id], onDelete: Cascade)
  technicianId String
  amount       Float
  note         String
  receiptUrl   String?
  status       String    @default("pending") // pending, paid
  paidAt       DateTime?
  createdAt    DateTime  @default(now())
}

model HisaabSettlement {
  id              String    @id @default(uuid())
  jobId           String
  job             Job       @relation(fields: [jobId], references: [id], onDelete: Cascade)
  technicianId    String
  amountExpected  Float
  amountCollected Float
  isFull          Boolean   @default(false)
  balanceDue      Float     @default(0)
  nextVisitDate   DateTime?
  discountAmount  Float?
  settledBy       String
  settledAt       DateTime  @default(now())
}

model TechnicianLedgerEntry {
  id           String   @id @default(uuid())
  technicianId String
  type         String   // advance, hisaab_given, expense_owed, expense_paid
  amount       Float
