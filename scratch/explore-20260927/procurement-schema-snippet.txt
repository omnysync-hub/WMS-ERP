model PurchaseRequisition {
  id              String                    @id @default(uuid())
  prNumber        String                    @unique
  requestedBy     String
  department      String?                   @default("HVAC Operations")
  site            String?
  dateRequired    DateTime?
  costCenter      String?
  projectCode     String?
  budgetCode      String?
  priority        String                    @default("Normal") // Normal, Urgent
  status          String                    @default("draft") // draft, submitted, approved, rejected, converted_to_po, partially_converted
  notes           String?
  attachments     String?                   // Specifications, drawings notes or URLs
  approvedBy      String?
  approvedAt      DateTime?
  rejectedBy      String?
  rejectedAt      DateTime?
  rejectionReason String?
  items           PurchaseRequisitionItem[]
  purchaseOrders  PurchaseOrder[]
  rfqLinks        RfqRequisitionLink[]
  createdAt       DateTime                  @default(now())
  updatedAt       DateTime                  @default(now()) @updatedAt
}

model PurchaseRequisitionItem {
  id                String              @id @default(uuid())
  prId              String
  pr                PurchaseRequisition @relation(fields: [prId], references: [id], onDelete: Cascade)
  productId         String?
  product           Product?            @relation(fields: [productId], references: [id])
  itemCode          String?
  description       String?
  quantity          Float
  unit              String              @default("unit")
  estimatedPrice    Float               @default(0)
  convertedQuantity Float               @default(0)
}

model RequestForQuotation {
  id             String                @id @default(uuid())
  rfqNumber      String                @unique
  title          String
  dueDate        DateTime
  status         String                @default("draft") // draft, sent, quotes_received, evaluated, awarded, closed
  notes          String?
  winnerVendorId String?
  awardedAt      DateTime?
  items          RfqItem[]
  vendors        RfqVendor[]
  prLinks        RfqRequisitionLink[]
  purchaseOrders PurchaseOrder[]
  createdAt      DateTime              @default(now())
  updatedAt      DateTime              @default(now()) @updatedAt
}

model RfqItem {
  id             String                @id @default(uuid())
  rfqId          String
  rfq            RequestForQuotation   @relation(fields: [rfqId], references: [id], onDelete: Cascade)
  productId      String?
  product        Product?              @relation(fields: [productId], references: [id])
  itemCode       String?
  description    String
  quantity       Float
  unit           String                @default("unit")
  targetPrice    Float?
  quotationItems VendorQuotationItem[]
}

model RfqVendor {
  id                 String                @id @default(uuid())
  rfqId              String
  rfq                RequestForQuotation   @relation(fields: [rfqId], references: [id], onDelete: Cascade)
  vendorId           String
  vendor             Vendor                @relation(fields: [vendorId], references: [id], onDelete: Cascade)
  status             String                @default("invited") // invited, quoted, declined, winner
  deliveryDays       Int?
  paymentTerms       String?
  qualityScore       Float?                @default(85)
  commercialScore    Float?
  totalQuoted        Float?                @default(0)
  isWinner           Boolean               @default(false)
  remarks            String?
  quotationReference String?
  submittedAt        DateTime?
  quotationItems     VendorQuotationItem[]
}

model VendorQuotationItem {
  id          String    @id @default(uuid())
  rfqVendorId String
  rfqVendor   RfqVendor @relation(fields: [rfqVendorId], references: [id], onDelete: Cascade)
  rfqItemId   String
  rfqItem     RfqItem   @relation(fields: [rfqItemId], references: [id], onDelete: Cascade)
  unitPrice   Float
  taxRate     Float     @default(0)
  lineTotal   Float
  notes       String?
}

model RfqRequisitionLink {
  id    String              @id @default(uuid())
  rfqId String
  rfq   RequestForQuotation @relation(fields: [rfqId], references: [id], onDelete: Cascade)
  prId  String
  pr    PurchaseRequisition @relation(fields: [prId], references: [id], onDelete: Cascade)

  @@unique([rfqId, prId])
}

model PurchaseOrder {
  id                 String              @id @default(uuid())
  poNumber           String              @unique
  poType             String              @default("standard") // standard, blanket, service
  prId               String?
  pr                 PurchaseRequisition? @relation(fields: [prId], references: [id])
  rfqId              String?
  rfq                RequestForQuotation? @relation(fields: [rfqId], references: [id])
  vendorId           String?
  vendor             Vendor?             @relation(fields: [vendorId], references: [id])
  supplierName       String
  supplierEmail      String
  poDate             DateTime            @default(now())
  expectedDeliveryDate DateTime?
  paymentTerms       String              @default("Net 30")
  currency           String              @default("PKR")
  shippingAddress    String?
  termsAndConditions String?
  status             String              @default("draft") // draft, approved, sent_to_vendor, partially_received, fully_received, closed, cancelled
  approvedBy         String?
  approvedAt         DateTime?
  sentAt             DateTime?
  totalAmount        Float               @default(0)
  whtAmount          Float               @default(0)
  netPayable         Float               @default(0)
  items              PurchaseOrderItem[]
  goodsReceipts      GoodsReceipt[]
  supplierInvoices   SupplierInvoice[]
  createdAt          DateTime            @default(now())
  updatedAt          DateTime            @default(now()) @updatedAt
}

model PurchaseOrderItem {
  id               String               @id @default(uuid())
  poId             String
  po               PurchaseOrder        @relation(fields: [poId], references: [id], onDelete: Cascade)
  productId        String?
  product          Product?             @relation(fields: [productId], references: [id])
  itemCode         String?
  description      String?
  unit             String               @default("unit")
  quantity         Float
  unitCost         Float
  discountPercent  Float                @default(0)
  taxPercent       Float                @default(0)
  lineTotal        Float                @default(0)
  deliverySchedule String?
  prItemId         String?
  quantityReceived Float                @default(0)
  quantityInvoiced Float                @default(0)
  grnItems         GoodsReceiptItem[]
  invoiceItems     SupplierInvoiceItem[]
}

model GoodsReceipt {
  id                 String              @id @default(uuid())
  grnNumber          String              @unique
  poId               String
  po                 PurchaseOrder       @relation(fields: [poId], references: [id])
  receivedBy         String
  receivedDate       DateTime            @default(now())
  warehouseLocation  String              @default("Central Warehouse")
  deliveryChallan    String?
  qualityStatus      String              @default("Accepted") // Accepted, Rejected, Hold
  status             String              @default("received")
  notes              String?
  accountingJournalId String?
  items              GoodsReceiptItem[]
  supplierInvoices   SupplierInvoice[]
  createdAt          DateTime            @default(now())
}

model GoodsReceiptItem {
  id               String               @id @default(uuid())
  grnId            String
  grn              GoodsReceipt         @relation(fields: [grnId], references: [id], onDelete: Cascade)
  poItemId         String?
  poItem           PurchaseOrderItem?   @relation(fields: [poItemId], references: [id])
  productId        String?
  product          Product?             @relation(fields: [productId], references: [id])
  itemCode         String?
  description      String?
  unit             String               @default("unit")
  quantityReceived Float
  quantityAccepted Float                @default(0)
  quantityRejected Float                @default(0)
  qualityStatus    String               @default("Accepted")
  rejectionReason  String?
  batchNumber      String?
  serialNumber     String?
  expiryDate       DateTime?
  invoiceItems     SupplierInvoiceItem[]
}

model SupplierInvoice {
  id               String                @id @default(uuid())
  invoiceNumber    String
  referenceNumber  String                @unique
  vendorId         String
  vendor           Vendor                @relation(fields: [vendorId], references: [id])
  poId             String?
  po               PurchaseOrder?        @relation(fields: [poId], references: [id])
  grnId            String?
  grn              GoodsReceipt?         @relation(fields: [grnId], references: [id])
  invoiceDate      DateTime              @default(now())
  dueDate          DateTime?
  subtotal         Float                 @default(0)
  taxAmount        Float                 @default(0)
  totalAmount      Float                 @default(0)
  matchStatus      String                @default("pending_match") // pending_match, matched, discrepancy, approved_for_payment, paid
  priceVariance    Float                 @default(0)
  quantityVariance Float                 @default(0)
  matchNotes       String?
  approvedBy       String?
  approvedAt       DateTime?
  paidAmount       Float                 @default(0)
  paymentStatus    String                @default("unpaid") // unpaid, partially_paid, paid
  journalEntryId   String?
  items            SupplierInvoiceItem[]
  payments         SupplierPayment[]
  createdAt        DateTime              @default(now())
}

model SupplierInvoiceItem {
  id              String             @id @default(uuid())
  invoiceId       String
  invoice         SupplierInvoice    @relation(fields: [invoiceId], references: [id], onDelete: Cascade)
  poItemId        String?
  poItem          PurchaseOrderItem? @relation(fields: [poItemId], references: [id])
  grnItemId       String?
  grnItem         GoodsReceiptItem?  @relation(fields: [grnItemId], references: [id])
  productId       String?
  product         Product?           @relation(fields: [productId], references: [id])
  description     String
  billedQuantity  Float
  billedUnitPrice Float
  poUnitPrice     Float?
  grnQuantity     Float?
  lineTotal       Float
  variance        Float              @default(0)
}
